\item El factor 50 representa prototipo/modelo; invertirlo antes de tomar la raíz.

\[\fitmath{\mathrm{Vm}=\frac{\mathrm{Vp}}{\sqrt{\mathrm{lam}}}}\]

\[\fitmath{\mathrm{Vm}=\frac{(10)}{\sqrt{(50)}}\approx 1{,}414214\,\mathrm{m/s}}\]

\end{enumerate}
