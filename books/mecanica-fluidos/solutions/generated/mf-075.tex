\begin{enumerate}

\item El resultado nulo es compatible con continuidad incompresible; no demuestra por sí solo que se satisfaga el balance de momento.

\[\fitmath{\mathrm{div}=a+b+c}\]

\[\fitmath{\mathrm{div}=(2)+(-3)+(1)\approx 0\,\mathrm{1/s}}\]

\end{enumerate}
