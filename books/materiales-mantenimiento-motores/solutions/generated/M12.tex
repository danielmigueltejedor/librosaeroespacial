\begin{enumerate}

\item Primer alcance: solo reparación activa.

\[\fitmath{\mathrm{Ai}=\frac{800}{800+4}\approx 0.9950249\,}\]

\item Segundo alcance: añadir la espera sin cambiar el numerador.

\[\fitmath{\mathrm{Aextended}=\frac{800}{800+4+20}\approx 0.9708738\,}\]

\end{enumerate}
