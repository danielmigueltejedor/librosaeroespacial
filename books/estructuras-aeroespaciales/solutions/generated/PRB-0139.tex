\begin{enumerate}

\item $\varepsilon_x = (\sigma_x - \nu\sigma_y)/E$ y $\varepsilon_y = (\sigma_y - \nu\sigma_x)/E$.

\item $\varepsilon_z = -\nu(\sigma_x + \sigma_y)/E$ porque $\sigma_z = 0$.

\end{enumerate}
