\begin{enumerate}

\item Compatibilidad: $f_1\,[(s_1 + s_{12})/A_1 + s_{12}/A_2] = f_2\,[(s_2 + s_{12})/A_2 + s_{12}/A_1]$, con $s_1 = 300$, $s_2 = 220$ y $s_{12} = 100$ mm los contornos exteriores y la pared común. Sea $a = (s_1 + s_{12})/A_1 + s_{12}/A_2$ y $b = (s_2 + s_{12})/A_2 + s_{12}/A_1$.

\item Entonces $f_2 = f_1\,a/b$, y el equilibrio $M_x = 2A_1 f_1 + 2A_2 f_2$ da $f_1$.

\item La tensión máxima es el mayor flujo dividido entre el espesor; la pared común soporta $f_1 - f_2$.

\end{enumerate}
