\[\fitmath{\mathrm{Pow}=\mathrm{em}\cdot \mathrm{md}\cdot 2\cdot \left(1-\mathrm{rea}\right)\cdot {\mathrm{uu}}^{2}}\]

\[\fitmath{\mathrm{Pow}=(0.975)\cdot (26)\cdot 2\cdot \left(1-(0.52)\right)\cdot {(322)}^{2}\approx 2523254\,\mathrm{W}}\]
