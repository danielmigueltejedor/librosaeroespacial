\item Impulso total.

\[\fitmath{I=F\cdot 38}\]

\[\fitmath{I=(17579.71047)\cdot 38\approx 668029\,\mathrm{N\,s}}\]

\item Velocidad ideal.

\[\fitmath{\mathrm{dv}=262\cdot 9.80665\cdot \ln\left(\frac{260+175}{175}\right)\approx 2339.54\,\mathrm{m/s}}\]

\item Relación empuje-peso con la masa inicial.

\[\fitmath{\mathrm{tw}=\frac{F}{\left(260+175\right)\cdot 9.80665}}\]

\[\fitmath{\mathrm{tw}=\frac{(17579.71047)}{\left(260+175\right)\cdot 9.80665}\approx 4.120992\,}\]

\end{enumerate}
