\item Empuje de un propulsor.

\[\fitmath{F_{1}=\frac{2\cdot 0.65\cdot 2500}{30000}\approx 0.1083333\,\mathrm{N}}\]

\item Número de propulsores necesario sin redondear.

\[\fitmath{\mathrm{nex}=\frac{0.3}{F_{1}}}\]

\[\fitmath{\mathrm{nex}=\frac{0.3}{(0.1083333333)}\approx 2.769231\,}\]

\item Empuje con tres propulsores.

\[\fitmath{F_{3}=3\cdot F_{1}}\]

\[\fitmath{F_{3}=3\cdot (0.1083333333)\approx 0.325\,\mathrm{N}}\]

\item Potencia total.

\[\fitmath{P=3\cdot 2500\approx 7500\,\mathrm{W}}\]

\item Gasto con tres propulsores.

\[\fitmath{\mathrm{md}=\frac{F_{3}}{30000}}\]

\[\fitmath{\mathrm{md}=\frac{(0.325)}{30000}\approx 1.083333\times10^{-5}\,\mathrm{kg/s}}\]

\item Área de panel.

\[\fitmath{A=\frac{P}{150}}\]

\[\fitmath{A=\frac{(7500)}{150}\approx 50\,\mathrm{m^{2}}}\]

\end{enumerate}
