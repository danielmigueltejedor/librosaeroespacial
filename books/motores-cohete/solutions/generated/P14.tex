\begin{enumerate}

\item Superficie de combustión constante.

\[\fitmath{\mathrm{Ab}=\frac{\pi\cdot {0.3}^{2}}{4}\approx 0.07068583\,\mathrm{m^{2}}}\]

\item Duración del quemado.

\[\fitmath{\mathrm{tb}=\frac{1.1}{0.0072}\approx 152.7778\,\mathrm{s}}\]

\item Gasto másico.

\[\fitmath{\mathrm{md}=1780\cdot \mathrm{Ab}\cdot 0.0072}\]

\[\fitmath{\mathrm{md}=1780\cdot (0.07068583471)\cdot 0.0072\approx 0.9059097\,\mathrm{kg/s}}\]

\item Empuje con impulso específico dado.

\[\fitmath{F=240\cdot 9.80665\cdot \mathrm{md}}\]

\[\fitmath{F=240\cdot 9.80665\cdot (0.9059096576)\approx 2132.145\,\mathrm{N}}\]

\item Masa de propulsante.

\[\fitmath{\mathrm{mp}=1780\cdot \mathrm{Ab}\cdot 1.1}\]

\[\fitmath{\mathrm{mp}=1780\cdot (0.07068583471)\cdot 1.1\approx 138.4029\,\mathrm{kg}}\]

\item Impulso total.

\[\fitmath{I=F\cdot \mathrm{tb}}\]

\[\fitmath{I=(2132.145346)\cdot (152.7777778)\approx 325744.4\,\mathrm{N\,s}}\]

\item Comprobación del impulso específico.

\[\fitmath{\mathrm{Ispc}=\frac{I}{\mathrm{mp}\cdot 9.80665}}\]

\[\fitmath{\mathrm{Ispc}=\frac{(325744.4279)}{(138.4028644)\cdot 9.80665}\approx 240\,\mathrm{s}}\]

\item Velocidad ideal de la ecuación del cohete.

\[\fitmath{\mathrm{dv}=240\cdot 9.80665\cdot \ln\left(\frac{\mathrm{mp}+90}{90}\right)}\]

\[\fitmath{\mathrm{dv}=240\cdot 9.80665\cdot \ln\left(\frac{(138.4028644)+90}{90}\right)\approx 2191.907\,\mathrm{m/s}}\]

\end{enumerate}
