\begin{enumerate}

\item Cociente de áreas de garganta.

\item Cociente de presiones final e inicial.

\[\fitmath{\mathrm{pr}={\left(\frac{1}{\tau}\right)}^{\frac{1}{1-0.32}}}\]

\[\fitmath{\mathrm{pr}={\left(\frac{1}{(1.1)}\right)}^{\frac{1}{1-0.32}}\approx 0.8692174\,}\]

\item Cociente de empuje y de gasto con CF constante.

\[\fitmath{\mathrm{fr}=\mathrm{pr}\cdot \tau}\]

\[\fitmath{\mathrm{fr}=(0.8692173855)\cdot (1.1)\approx 0.9561391\,}\]

\item Pérdida relativa de empuje en por ciento.

\[\fitmath{\mathrm{loss}=\left(1-\mathrm{fr}\right)\cdot 100}\]

\[\fitmath{\mathrm{loss}=\left(1-(0.956139124)\right)\cdot 100\approx 4.386088\,}\]

\end{enumerate}
