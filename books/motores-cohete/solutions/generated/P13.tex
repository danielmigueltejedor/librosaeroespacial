\begin{enumerate}

\item Cociente de áreas de combustión y garganta.

\[\fitmath{\mathrm{Kn}=\frac{0.72}{0.003}\approx 240\,}\]

\item Parámetro adimensional del equilibrio.

\[\fitmath{w=\frac{1750\cdot 0.008\cdot 1520\cdot \mathrm{Kn}}{6900000}}\]

\[\fitmath{w=\frac{1750\cdot 0.008\cdot 1520\cdot (240)}{6900000}\approx 0.7401739\,}\]

\item Presión de cámara de equilibrio.

\[\fitmath{\mathrm{pc}=6900000\cdot {w}^{\frac{1}{1-0.35}}}\]

\[\fitmath{\mathrm{pc}=6900000\cdot {(0.740173913)}^{\frac{1}{1-0.35}}\approx 4343343\,\mathrm{Pa}}\]

\item Velocidad de regresión a esa presión.

\[\fitmath{r=0.008\cdot {\left(\frac{\mathrm{pc}}{6900000}\right)}^{0.35}}\]

\[\fitmath{r=0.008\cdot {\left(\frac{(4343343.01)}{6900000}\right)}^{0.35}\approx 0.006803482\,\mathrm{m/s}}\]

\item Gasto generado por la superficie.

\[\fitmath{m_{1}=1750\cdot 0.72\cdot r}\]

\[\fitmath{m_{1}=1750\cdot 0.72\cdot (0.006803482159)\approx 8.572388\,\mathrm{kg/s}}\]

\item Gasto descargado por la garganta.

\[\fitmath{m_{2}=\frac{\mathrm{pc}\cdot 0.003}{1520}}\]

\[\fitmath{m_{2}=\frac{(4343343.01)\cdot 0.003}{1520}\approx 8.572388\,\mathrm{kg/s}}\]

\end{enumerate}
