\begin{enumerate}

\item Velocidad de escape.

\[\fitmath{\mathrm{ve}=1800\cdot 9.80665\approx 17651.97\,\mathrm{m/s}}\]

\item Propulsante por la ecuación del cohete.

\[\fitmath{\mathrm{mp}=900\cdot \left(1-\exp\left(\frac{-450}{\mathrm{ve}}\right)\right)}\]

\[\fitmath{\mathrm{mp}=900\cdot \left(1-\exp\left(\frac{-450}{(17651.97)}\right)\right)\approx 22.65363\,\mathrm{kg}}\]

\item Empuje de potencia de chorro.

\[\fitmath{F=\frac{2\cdot 0.55\cdot 1350}{\mathrm{ve}}}\]

\[\fitmath{F=\frac{2\cdot 0.55\cdot 1350}{(17651.97)}\approx 0.08412659\,\mathrm{N}}\]

\item Gasto.

\[\fitmath{\mathrm{md}=\frac{F}{\mathrm{ve}}}\]

\[\fitmath{\mathrm{md}=\frac{(0.08412658757)}{(17651.97)}\approx 4.765847\times10^{-6}\,\mathrm{kg/s}}\]

\item Tiempo de maniobra.

\[\fitmath{t=\frac{\mathrm{mp}}{\mathrm{md}}}\]

\[\fitmath{t=\frac{(22.65363447)}{(4.76584696\times10^{-6})}\approx 4753328\,\mathrm{s}}\]

\item Tiempo en días.

\[\fitmath{\mathrm{dias}=\frac{t}{86400}}\]

\[\fitmath{\mathrm{dias}=\frac{(4753328.139)}{86400}\approx 55.01537\,\mathrm{d}}\]

\end{enumerate}
