\item Razón de masas con carga útil.

\[\fitmath{R=\frac{420+70+110}{70+110}\approx 3.333333\,}\]

\item Velocidad ideal nominal.

\[\fitmath{\mathrm{dv}=245\cdot 9.80665\cdot \ln\left(R\right)}\]

\[\fitmath{\mathrm{dv}=245\cdot 9.80665\cdot \ln\left((3.333333333)\right)\approx 2892.7\,\mathrm{m/s}}\]

\item Velocidad ideal con impulso específico reducido.

\[\fitmath{\mathrm{dv}_{2}=0.97\cdot 245\cdot 9.80665\cdot \ln\left(R\right)}\]

\[\fitmath{\mathrm{dv}_{2}=0.97\cdot 245\cdot 9.80665\cdot \ln\left((3.333333333)\right)\approx 2805.919\,\mathrm{m/s}}\]

\item Pérdida por la reducción de impulso específico.

\[\fitmath{\mathrm{loss}=\mathrm{dv}-\mathrm{dv}_{2}}\]

\[\fitmath{\mathrm{loss}=(2892.700276)-(2805.919268)\approx 86.78101\,\mathrm{m/s}}\]
