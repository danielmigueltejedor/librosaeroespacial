\par\textbf{Procedimiento completo para comparar tu solución.} El equilibrio exige $L=mg$, $T=D$. Primero hallar $q$ y peso; después $C_L=W/(qS)$, resistencia $D=qSC_D$ y potencia al eje $P_e=DV/\eta_p$.

\begin{enumerate}

\item Calcular presión dinámica a la velocidad prescrita.

\[\fitmath{q=\frac{1.06\cdot {55}^{2}}{2}\approx 1603.25\,\mathrm{Pa}}\]

\item Convertir masa a peso.

\[\fitmath{W=920\cdot 9.80665\approx 9022.118\,\mathrm{N}}\]

\item Despejar coeficiente de sustentación.

\[\fitmath{C_L=\frac{W}{q\cdot 16}}\]

\[\fitmath{C_L=\frac{(9022.118)}{(1603.25)\cdot 16}\approx 0.3517121\,}\]

\item Convertir coeficiente de resistencia en fuerza.

\[\fitmath{D=q\cdot 16\cdot 0.038}\]

\[\fitmath{D=(1603.25)\cdot 16\cdot 0.038\approx 974.776\,\mathrm{N}}\]

\item Potencia útil contra resistencia.

\[\fitmath{\mathrm{Pu}=D\cdot 55}\]

\[\fitmath{\mathrm{Pu}=(974.776)\cdot 55\approx 53612.68\,\mathrm{W}}\]

\item Dividir por rendimiento para la potencia al eje.

\[\fitmath{\mathrm{Pe}=\frac{\mathrm{Pu}}{0.78}}\]

\[\fitmath{\mathrm{Pe}=\frac{(53612.68)}{0.78}\approx 68734.21\,\mathrm{W}}\]

\end{enumerate}
