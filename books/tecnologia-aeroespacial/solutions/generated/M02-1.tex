\begin{enumerate}

\item Convertir kW a W y dividir por masa de A.

\[\fitmath{A=\frac{10000}{260}\approx 38.46154\,\mathrm{W/kg}}\]

\item Repetir para B con sus propios datos.

\[\fitmath{B=\frac{18000}{310}\approx 58.06452\,\mathrm{W/kg}}\]

\end{enumerate}
