\par\textbf{Procedimiento completo para comparar tu solución.} En cada estado, dividir momento de masa total entre masa total. Consumir elimina masa y su momento; transferir mantiene masa total y altera momento en $\Delta m(x_{destino}-x_{origen})$.

\begin{enumerate}

\item Masa inicial.

\[\fitmath{M0=2000+200+300\approx 2500\,\mathrm{kg}}\]

\item Momento inicial respecto al datum.

\[\fitmath{S0=2000\cdot 2.5+200\cdot 1+300\cdot 4\approx 6400\,\mathrm{kg\,m}}\]

\item Centro inicial.

\[\fitmath{x0=\frac{S0}{M0}}\]

\[\fitmath{x0=\frac{(6400)}{(2500)}\approx 2.56\,\mathrm{m}}\]

\item Masa tras consumo delantero.

\[\fitmath{M1=M0-100}\]

\[\fitmath{M1=(2500)-100\approx 2400\,\mathrm{kg}}\]

\item Restar el momento de la masa consumida.

\[\fitmath{S1=S0-100\cdot 1}\]

\[\fitmath{S1=(6400)-100\cdot 1\approx 6300\,\mathrm{kg\,m}}\]

\item Centro después del consumo.

\[\fitmath{x1=\frac{S1}{M1}}\]

\[\fitmath{x1=\frac{(6300)}{(2400)}\approx 2.625\,\mathrm{m}}\]

\item Trasladar 40 kg desde el depósito trasero al delantero.

\[\fitmath{S2=S1+40\cdot \left(1-4\right)}\]

\[\fitmath{S2=(6300)+40\cdot \left(1-4\right)\approx 6180\,\mathrm{kg\,m}}\]

\item La transferencia conserva la masa total.

\[\fitmath{x2=\frac{S2}{M1}}\]

\[\fitmath{x2=\frac{(6180)}{(2400)}\approx 2.575\,\mathrm{m}}\]

\end{enumerate}
